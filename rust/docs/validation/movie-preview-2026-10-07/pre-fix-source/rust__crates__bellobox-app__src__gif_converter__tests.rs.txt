use super::model::{Model, Source};
use bellobox_core::recording::gif::{GifError, GifExportResult, ReplacePolicy};
use std::path::PathBuf;

fn ready() -> Model {
    let mut model = Model::default();
    let source = Source::Synthetic;
    let (generation, cancellation) = model.load(source.clone()).unwrap();
    model.loaded(generation, source.inspect(cancellation));
    model
}
fn destination() -> PathBuf {
    let nonce = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap()
        .as_nanos();
    std::env::temp_dir().join(format!(
        "bellobox-gif-test-{}-{nonce}.gif",
        std::process::id()
    ))
}
#[test]
fn source_change_resets_trim_and_result_but_keeps_format() {
    let mut model = ready();
    model.options.frames_per_second = 10;
    model.options.max_width = 320;
    model.options.loops = false;
    model.options.trim_start = 1.;
    let (generation, _) = model.load(Source::Synthetic).unwrap();
    assert_eq!(model.options.trim_start, 0.);
    assert_eq!(model.options.trim_end, None);
    assert_eq!(model.options.frames_per_second, 10);
    assert_eq!(model.options.max_width, 320);
    assert!(!model.options.loops);
    model.loaded(generation, Source::Synthetic.inspect(Default::default()));
    assert_eq!(model.options.trim_end, Some(3.));
}

#[test]
fn busy_rejects_source_replacement_and_duplicate_export() {
    let mut model = ready();
    let work = model
        .begin(destination(), ReplacePolicy::RefuseExisting)
        .unwrap();
    assert!(model.load(Source::Synthetic).is_none());
    assert!(
        model
            .begin(destination(), ReplacePolicy::RefuseExisting)
            .is_none()
    );
    model.cancel();
    assert!(
        model.busy(),
        "cancel does not admit new work until old worker settles"
    );
    let generation = work.generation;
    model.finished(generation, work.run());
    assert!(!model.busy());
    assert_eq!(model.source, Some(Source::Synthetic));
}
#[test]
fn actual_synthetic_export_publishes_then_reports_result() {
    let mut model = ready();
    model.options.trim_start = 0.5;
    model.options.trim_end = Some(0.7);
    model.options.loops = false;
    let path = destination();
    let work = model
        .begin(path.clone(), ReplacePolicy::RefuseExisting)
        .unwrap();
    let generation = work.generation;
    let result = work.run().unwrap();
    assert_eq!(result.frame_count, 3);
    assert_eq!(result.size, (320, 180));
    assert!((result.duration - 0.2).abs() < 1e-6);
    let bytes = std::fs::read(&path).unwrap();
    assert!(bytes.starts_with(b"GIF"));
    assert!(!bytes.windows(11).any(|s| s == b"NETSCAPE2.0"));
    model.finished(generation, Ok(result));
    assert_eq!(model.result_revision, 1);
    assert_eq!(model.source, Some(Source::Synthetic));
    assert!(model.status.starts_with("Saved"));
    std::fs::remove_file(path).unwrap();
}

#[test]
fn cancelled_retry_retains_last_successful_result() {
    let mut model = ready();
    let work = model
        .begin(destination(), ReplacePolicy::RefuseExisting)
        .unwrap();
    model.finished(
        work.generation,
        Ok(GifExportResult {
            path: "prior.gif".into(),
            frame_count: 1,
            size: (1, 1),
            duration: 0.1,
            file_size: 30,
        }),
    );
    drop(work);
    let retry = model
        .begin(destination(), ReplacePolicy::RefuseExisting)
        .unwrap();
    model.cancel();
    model.finished(retry.generation, retry.run());
    assert_eq!(
        model.result.as_ref().unwrap().path,
        PathBuf::from("prior.gif")
    );
}

    let directory = tempfile::tempdir().unwrap();
    let target = directory.path().join("target");
    let link = directory.path().join("link.gif");
    std::fs::write(&target, b"not GIF").unwrap();
    symlink(&target, &link).unwrap();
    let mut result = GifExportResult {
        path: link,
        frame_count: 1,
        size: (1, 1),
        duration: 0.1,
        file_size: 7,

    assert!(super::model::open_preview(&result).is_err());
    let fifo = directory.path().join("pipe.gif");
    let name = std::ffi::CString::new(fifo.as_os_str().as_encoded_bytes()).unwrap();
    // SAFETY: the owned CString is NUL-terminated and alive for this call;
    // the path is inside this test's private disposable directory.

    result.path = fifo;
    result.file_size = 0;
    assert!(super::model::open_preview(&result).is_err());
}

#[test]
fn preview_repeat_pins_opened_file_when_path_is_replaced() {
    let mut model = ready();
    model.options.trim_end = Some(0.1);
    let path = destination();
    let work = model
        .begin(path.clone(), ReplacePolicy::RefuseExisting)
        .unwrap();
    let result = work.run().unwrap();
    let mut preview = super::model::open_preview(&result).unwrap();
    let original = preview.next_frame().unwrap().unwrap().png;
    while preview.next_frame().unwrap().is_some() {}
    let replacement = destination();
    let mut bytes = std::fs::read(&path).unwrap();
    let mut cursor = 13;
    if bytes[10] & 0x80 != 0 {
        cursor += 3 * (2usize << (bytes[10] & 7));
    }
    // Walk extension sub-blocks to the first full-frame local color table.
    while bytes[cursor] == 0x21 {
        cursor += 2;
        loop {
            let length = usize::from(bytes[cursor]);
            cursor += 1;
            if length == 0 {
                break;
            }
            cursor += length;
        }
    }
    assert_eq!(bytes[cursor], 0x2c);
    let packed = bytes[cursor + 9];
    assert_ne!(packed & 0x80, 0);
    let palette = cursor + 10;
    for channel in &mut bytes[palette..palette + 3 * (2usize << (packed & 7))] {
        *channel ^= 0xff;
    }
    std::fs::write(&replacement, bytes).unwrap();
    let mut replacement_result = result.clone();
    replacement_result.path = replacement.clone();
    let mut replacement_preview = super::model::open_preview(&replacement_result).unwrap();
    assert_ne!(
        replacement_preview.next_frame().unwrap().unwrap().png,
        original
    );
    std::fs::rename(&replacement, &path).unwrap();
    let mut repeated = preview.rewind().unwrap();
    assert_eq!(repeated.next_frame().unwrap().unwrap().png, original);
    std::fs::remove_file(path).unwrap();
}

#[gpui::test]
fn rapid_pause_play_preserves_every_prefetched_frame_exactly_once(cx: &mut gpui::TestAppContext) {
    let root = view(cx);
    let path = destination();
    root.update(cx, |view, cx| {
        view.model.options.trim_end = Some(0.3);
        view.model.options.loops = false;
        view.convert(path.clone(), cx);
    });
    cx.run_until_parked();
    let result = root.read_with(cx, |view, _| view.model.result.clone().unwrap());
    let mut reader = super::model::open_preview(&result).unwrap();
    let mut expected = Vec::new();
    while let Some(frame) = reader.next_frame().unwrap() {
        expected.push(frame);
    }
    let expected_pixels: Vec<Vec<u8>> = expected
        .iter()
        .map(|frame| {
            cx.update(|cx| {
                gpui::Image::from_bytes(gpui::ImageFormat::Png, frame.png.clone())
                    .to_image_data(cx.svg_renderer())
                    .unwrap()
                    .as_bytes(0)
                    .unwrap()
                    .to_vec()
            })
        })
        .collect();
    root.read_with(cx, |view, _| {
        assert_eq!(
            view.preview.image.as_ref().unwrap().as_bytes(0).unwrap(),
            expected_pixels[0]
        )
    });
    root.update(cx, |view, cx| view.toggle_preview(cx));
    cx.run_until_parked();
    for (frame, pixels) in expected.iter().zip(expected_pixels.iter()).skip(1) {
        root.read_with(cx, |view, _| {
            assert_eq!(
                view.preview.image.as_ref().unwrap().as_bytes(0).unwrap(),
                pixels.as_slice()
            )
        });
        // Pending decode/deadline must survive repeated Pause/Play without an
        // extra decode or a discarded already-read frame.
        root.update(cx, |view, cx| {
            for _ in 0..3 {
                view.toggle_preview(cx);
                view.toggle_preview(cx);
            }
        });
        cx.run_until_parked();
        root.read_with(cx, |view, _| {
            assert_eq!(
                view.preview.image.as_ref().unwrap().as_bytes(0).unwrap(),
                pixels.as_slice()
            )
        });
        cx.executor()
            .advance_clock(std::time::Duration::from_millis(
                u64::from(frame.delay_centiseconds) * 10,
            ));
        cx.run_until_parked();
    }
    root.read_with(cx, |view, _| assert!(!view.preview.playing));
    std::fs::remove_file(path).unwrap();
}
#[gpui::test]
fn prepared_render_image_uses_gpui_bgra_and_opaque_export_alpha(cx: &mut gpui::TestAppContext) {
    let root = view(cx);
    let path = destination();
    root.update(cx, |view, cx| {
        view.model.options.trim_start = 0.5;
        view.model.options.trim_end = Some(0.6);
        view.convert(path.clone(), cx);
    });
    cx.run_until_parked();
    root.read_with(cx, |view, _| {
        let pixels = view.preview.image.as_ref().unwrap().as_bytes(0).unwrap();
        // GPUI0.2.2 Image::to_image_data swaps PNG RGBA's R/B exactly once.
        // The own GIF encoder's opaque output must retain alpha255.
        assert_eq!(&pixels[0..4], &[180, 130, 230, 255]);
        assert_eq!(&pixels[200 * 4..200 * 4 + 4], &[180, 130, 40, 255]);
        let bottom = (100 * 320 + 200) * 4;
        assert_eq!(&pixels[bottom..bottom + 4], &[180, 60, 40, 255]);
    });
